cmake_minimum_required(VERSION 3.21)
project(AccessibleWheel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick QuickControls2 WebEngineQuick QuickTest)
qt_add_executable(wheel_demo main.cpp)
target_link_libraries(wheel_demo PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::WebEngineQuick)
target_compile_definitions(wheel_demo PRIVATE DEMO_QML="${CMAKE_CURRENT_SOURCE_DIR}/qml/Demo.qml")
enable_testing()
qt_add_executable(wheel_tests tests/main.cpp)
target_link_libraries(wheel_tests PRIVATE Qt6::QuickTest Qt6::WebEngineQuick)
target_compile_definitions(wheel_tests PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
add_test(NAME native_wheel COMMAND wheel_tests -input ${CMAKE_CURRENT_SOURCE_DIR}/tests)
